⚡ Invoice Generator API

Send invoice data as JSON, get back a real PDF or Excel file. All fields are optional — unset ones auto-fill from RIDHI ENTERPRISE defaults.

🌐 http://localhost:3000
GET/apiFull API docs
GET/api/healthHealth check
GET/api/defaultsView all default field values
POST/api/invoice/pdfReturns a PDF file Main
POST/api/invoice/xlsxReturns an Excel (.xlsx) file Main
POST/api/invoice/jsonReturns calculated totals as JSON

🟦 JavaScript / fetch

const BASE = "http://localhost:3000"; // ── Download PDF ────────────────────────────── async function downloadInvoicePDF(invoiceData) { const res = await fetch(`${BASE}/api/invoice/pdf`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(invoiceData) }); const blob = await res.blob(); const url = URL.createObjectURL(blob); Object.assign(document.createElement("a"), { href: url, download: "invoice.pdf" }).click(); } // ── Download Excel ──────────────────────────── async function downloadInvoiceXLSX(invoiceData) { const res = await fetch(`${BASE}/api/invoice/xlsx`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(invoiceData) }); const blob = await res.blob(); const url = URL.createObjectURL(blob); Object.assign(document.createElement("a"), { href: url, download: "invoice.xlsx" }).click(); } // ── Usage ───────────────────────────────────── downloadInvoicePDF({ invoiceNo: "RE-39", invoiceDate: "01-04-2026", receiverName: "NEW CLIENT INDUSTRIES", period: "01/03/2026 to 31/03/2026", items: [ { desc: "Loading /Unloading", rate: 500, persons: 200, days: 1 } ] });

🐍 Python

import requests BASE = "http://localhost:3000" payload = { "invoiceNo": "RE-39", "invoiceDate": "01-04-2026", "receiverName": "NEW CLIENT INDUSTRIES", "items": [ { "desc": "Loading /Unloading", "rate": 500, "persons": 200, "days": 1 } ] } # Save PDF r = requests.post(f"{BASE}/api/invoice/pdf", json=payload) open("invoice.pdf", "wb").write(r.content) # Save Excel r = requests.post(f"{BASE}/api/invoice/xlsx", json=payload) open("invoice.xlsx", "wb").write(r.content) # Get JSON totals data = requests.post(f"{BASE}/api/invoice/json", json=payload).json() print(data["totals"]["grandTotal"]) # 1,15,500 print(data["totals"]["grandTotalWords"]) # One Lakh Fifteen Thousand...
